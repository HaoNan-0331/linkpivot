```typescript
async function executeSSH(host: string, port: number, username: string, password: string, command: string, timeout: number = 30000): Promise<string> {
  return new Promise((resolve, reject) => {
    const client = new Client()
    let settled = false
    const timer = setTimeout(() => {
      if (settled) return
      settled = true
      // timeout 兜底路径：end 可能因对端不响应 EOF 不回收，追加 destroy() 强制销毁 socket（D-6-2）
      try { client.end() } catch {}
      try { client.destroy() } catch {}
      reject(new Error(`SSH timeout after ${timeout}ms`))
    }, timeout)

    const done = (fn: () => void) => {
      if (settled) return
      settled = true
      fn()
    }

    try {
      client.on('ready', () => {
        client.exec(command, (err, stream) => {
          if (err) { done(() => reject(err)); return }
          let output = ''
          stream.on('data', (data: Buffer) => { output += data.toString() })
          stream.stderr.on('data', (data: Buffer) => { output += data.toString() })
          stream.on('close', () => { done(() => resolve(output)) })
          stream.on('error', (e: Error) => { done(() => reject(e)) })
        })
      })
      client.on('error', (err) => { done(() => reject(err)) })
      client.connect({
        host, port, username, password, readyTimeout: timeout,
        algorithms: {
          kex: ['curve25519-sha256', 'ecdh-sha2-nistp256', 'ecdh-sha2-nistp384', 'ecdh-sha2-nistp521', 'diffie-hellman-group-exchange-sha256', 'diffie-hellman-group14-sha256', 'diffie-hellman-group14-sha1', 'diffie-hellman-group1-sha1'],
          cipher: ['aes128-gcm', 'aes256-gcm', 'aes128-ctr', 'aes192-ctr', 'aes256-ctr', 'aes128-cbc', 'aes256-cbc', '3des-cbc'],
          serverHostKey: ['ssh-ed25519', 'ecdsa-sha2-nistp256', 'rsa-sha2-512', 'rsa-sha2-256', 'ssh-rsa'],
        },
      })
    } finally {
      // finally 保证任意 resolve/reject 路径均清 timer + end（D-6-2 锁定模式）
      // 注意：finally 在 executor 同步抛出时立即触发；但 ready/exec 回调内的 resolve/reject 是异步的，
      // 故 done() + settled-flag 与 finally 配合：finally 处理同步路径，settled-flag + 各回调内 done() 处理异步路径。
      // 为覆盖异步回调路径，在 done() 内也清 timer + end（双保险，幂等）。
    }
  })
}
```

注意：ssh2 是事件驱动，executor 同步 try/finally 的 finally 块在 `client.connect` 返回后立即执行（此时 ready/error 尚未回调）。因此 finally 仅能清同步异常路径；异步回调路径（ready/exec/stream）的资源回收必须依赖 `done()` 回调内统一清。最终落地形态须为：

- 抽 `cleanup()` 内部函数：`clearTimeout(timer); try { client.end() } catch {}`（幂等，多次调用无害——clearTimeout 已 fire id 无害；client.end() 已 end 后再 end 可能抛被 catch ignore）。
- timeout fire 路径：`cleanup()` 后追加 `try { client.destroy() } catch {}`（D-6-2：timeout 兜底 end+destroy 双调用）。
- ready/exec stream close → `cleanup(); done(resolve, output)`。
- stream error / exec err / client error → `cleanup(); done(reject, err)`。
- executor 体内不再保留散落的 `clearTimeout(timeoutId)` / `client.end()`，全部走 cleanup() 统一出口。

executor 在所有回调注册完成后立即 `return`（不抛同步异常则 finally 不需要在 executor 末尾再清；若 connect() 同步抛则 try/catch 包住走 cleanup + reject）。**planner 授权 executor 在「形态 a (Promise executor 内 cleanup 统一出口)」与「形态 b (async/await 包装 + 外层 try/finally)」间裁量**，但**模式锁定**：所有 resolve/reject/throw 出口均经过 cleanup（clearTimeout + client.end），timeout fire 路径额外 client.destroy()。

```typescript
async function executeTelnet(host: string, port: number, username: string, password: string, command: string, timeout: number = 30000): Promise<string> {
  const connection = new Telnet()
  // D-6-1：补自有 setTimeout 包 connect+exec 整体（库级 connect.timeout/execTimeout 不完全可靠，与 executeSSH 外层兜底对齐）
  let timer: NodeJS.Timeout | undefined
  let timedOut = false
  const result = await new Promise<string>((resolve, reject) => {
    timer = setTimeout(() => {
      timedOut = true
      // timeout 兜底：destroy 强制销毁 socket（telnet-client end 是优雅，destroy 强制）
      try { connection.destroy() } catch {}
      reject(new Error(`Telnet timeout after ${timeout}ms`))
    }, timeout)

    ;(async () => {
      try {
        await connection.connect({
          host, port, timeout, username, password,
          loginPrompt: /Username:|login:/i,
          passwordPrompt: /Password:/i,
          shellPrompt: /[>#]/,
          echoLines: 0, stripShellPrompt: true, execTimeout: timeout, newlineReplace: true,
        })
        const out = await connection.exec(command)
        resolve(out)
      } catch (err) {
        reject(err)
      }
    })()
  }).finally(() => {
    // D-6-2 finally：清 timer + end（优雅）+ destroy（强制，覆盖 timeout 已 destroy 场景的幂等）
    if (timer) clearTimeout(timer)
    try { connection.end() } catch {}
    if (timedOut) { try { connection.destroy() } catch {} }
  })
  return result
}
```

注意：telnet-client 的 end/destroy 是 async，finally 内调即可（D-6-2：不强 await，触发回收即可；executor 裁量是否 `await connection.end()`，但模式锁定为 finally 内 end+destroy）。`timedOut` 标记用于让 destroy 仅在 timeout 路径触发（与 SSH 的 timeout=end+destroy 对齐，正常路径仅 end）。
