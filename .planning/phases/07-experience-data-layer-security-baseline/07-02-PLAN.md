**2. electron/ipc/experienceIpc.ts**（新建，10 channel 全 secure 包装——经验属特权操作，全走 secure 不走 safe）:
```typescript
import { ipcMain } from 'electron'
import { ExperienceService, MAX_BATCH } from '../services/experienceService'
import type { ExperienceInput, ExperienceUpdateInput, ExperienceListInput } from '../../src/types/experience'
import { secure } from '../utils/authGuard'

export function registerExperienceIpc() {
  // 经验属特权操作（涉敏感 attrs/凭证片段），全 secure 包装（鉴权 + 异常脱敏）
  ipcMain.handle('experience:list', secure((_e, opts?: ExperienceListInput) =>
    ExperienceService.list(opts || {})))

  ipcMain.handle('experience:get', secure((_e, id: string) =>
    ExperienceService.getById(id)))

  ipcMain.handle('experience:create', secure((_e, input: ExperienceInput) =>
    ExperienceService.create(input)))

  ipcMain.handle('experience:update', secure((_e, id: string, fields: ExperienceUpdateInput) =>
    ExperienceService.update(id, fields)))

  ipcMain.handle('experience:delete', secure((_e, id: string) =>
    ExperienceService.delete(id)))

  ipcMain.handle('experience:invalidate', secure((_e, id: string) =>
    ExperienceService.invalidate(id)))

  ipcMain.handle('experience:relate-device', secure((_e, experienceId: string, deviceId: string, relationType?: string) =>
    ExperienceService.relateDevice(experienceId, deviceId, relationType)))

  ipcMain.handle('experience:unrelate-device', secure((_e, experienceId: string, deviceId: string) =>
    ExperienceService.unrelateDevice(experienceId, deviceId)))

  ipcMain.handle('experience:list-by-device', secure((_e, deviceId: string, includeInvalid?: boolean) =>
    ExperienceService.listByDevice(deviceId, includeInvalid)))

  ipcMain.handle('experience:list-devices', secure((_e, experienceId: string) =>
    ExperienceService.listDevicesByExperience(experienceId)))
}
```

**注意:** 全部 10 channel 用 `secure(...)`（鉴权 + 异常脱敏）。经验数据天然属登录后特权操作，无登录前 channel（不像 auth:* 需 safe）。MAX_BATCH 在 service 层强制（list 内 throw），IPC 层不重复校验，透传 opts.limit 即可——避免双层校验逻辑漂移。返回值不经 IPC 层二次脱敏（service 层 getById/list 已 decField 回填 attrs 并 delete attrs_enc，密文/凭证不外泄；listDevicesByExperience 返的 devices 原始行含 name_enc 等密文列——但本 phase 不在 IPC 层解密设备名，设备脱敏属 device 域既有契约，Phase 10 浏览页消费时复用 device:list 脱敏通道，本 plan 不越域处理）。

| Boundary | Description |
|----------|-------------|
| renderer → main（IPC 网关） | experience:* channel 是 renderer 访问经验的唯一入口；每个 handler 经 secure() 鉴权 + 异常脱敏 |
