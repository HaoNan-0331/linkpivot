    ```typescript
    // FE-04 (D-5-6): 模块级 LRU 缓存（非 per-instance），ChunkContent 频繁 re-mount 时跨实例复用，
    // 存活卸载。keyed by file_path → base64 data url。LRU 按 count 有界。
    // FE-04 (D-5-5): in-flight 去重 Map，同图并发请求复用同一 Promise（chunk 搜索切换/编辑去重）。
    // 客户端 AbortController：better-sqlite3 同步读不可真中断，AbortController 落地为
    // 结构化取消标志 + 卸载防 setState + 配合 in-flight 去重。

    const CACHE_MAX_ENTRIES = 100  // LRU 容量阈值（Claude's Discretion，D-5-6 count vs bytes 选 count）

    const cache = new Map<string, string>()                  // LRU（Map 保持插入顺序，淘汰最老）
    const inFlight = new Map<string, Promise<string>>()      // in-flight 去重

    /**
     * 取 KB 图片数据。优先读缓存 → 复用 in-flight → 否则调 IPC 并入缓存。
     * signal.aborted 时调用方在 then 前判 signal.aborted 决定是否 setState（本函数仍返回 Promise）。
     */
    export async function getImage(path: string, signal: AbortSignal): Promise<string> {
      // 1. 缓存命中
      const cached = cache.get(path)
      if (cached !== undefined) return cached

      // 2. in-flight 复用（同图并发请求）
      const existing = inFlight.get(path)
      if (existing) return existing

      // 3. 发起新请求
      const p = (async () => {
        try {
          const data = await window.api.kb.getImageData(path)
          if (data) {
            // LRU 淘汰：超容量删最老（Map 迭代顺序 = 插入顺序）
            if (cache.size >= CACHE_MAX_ENTRIES) {
              const oldest = cache.keys().next().value
              if (oldest !== undefined) cache.delete(oldest)
            }
            cache.set(path, data)
            return data
          }
          throw new Error('图片数据为空')
        } finally {
          inFlight.delete(path)   // 请求结束（成功/失败）清 in-flight
        }
      })()
      inFlight.set(path, p)

      // 4. 支持 abort：signal.aborted 后续 setState 由调用方守卫
      const onAbort = () => { /* 标志位由调用方读 signal.aborted，共享请求不因单方 abort 取消 */ }
      if (signal.aborted) onAbort()
      else signal.addEventListener('abort', onAbort, { once: true })

      return p
    }

    /** 测试/切换场景可手动清缓存（可选 export，Claude's Discretion） */
    export function clearImageCache(): void { cache.clear(); inFlight.clear() }
    ```
    > **设计要点：** in-flight 共享 Promise 不因单个调用方 abort 而取消（其他调用方仍待结果）；abort 语义由调用方在 `.then` 前判 `signal.aborted` 决定是否 setState（见 ChunkContent 改造）。LRU 用 Map 插入顺序淘汰最老（O(1)）。cache 命中返回 `data !== undefined` 判定（空字符串图片数据理论不存在，getImageData 返回 string|null）。
